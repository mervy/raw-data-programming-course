Q6

Divisão inteira e conversões

Sem executar, diga o que o programa imprime. Depois execute e explique cada linha.

#include <stdio.h>

int main(void) {
    int a = 7, b = 2;
    printf("%d\n", a / b);
    printf("%d\n", -7 / 2);
    printf("%d\n", -7 % 2);
    printf("%.2f\n", a / b * 1.0);
    printf("%.2f\n", (double) a / b);
    unsigned int u = 0;
    printf("%u\n", u - 1);
    return 0;
}

Achando o vazamento

O programa abaixo vaza memória. Encontre os dois vazamentos, corrija e diga qual ferramenta do curso confirmaria a correção.

#include <stdlib.h>
#include <string.h>

char *saudacao(const char *nome) {
    char *s = malloc(strlen(nome) + 7);
    strcpy(s, "Olá, ");
    strcat(s, nome);
    return s;
}

int main(void) {
    for (int i = 0; i < 3; i++) {
        char *msg = saudacao("Ana");
        msg = saudacao("Bia");
        free(msg);
    }
    return 0;
}
